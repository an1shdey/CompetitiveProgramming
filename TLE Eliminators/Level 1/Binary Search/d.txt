class Solution {
public:
    int sumOddLengthSubarrays(vector<int>& arr) {
        int sum=0;
        for(int i=0;i<arr.size();i++)
        {
            int temp_sum=0;
            for(int j=i;j<arr.size();j++)
            {
                temp_sum+=arr[j];
                if((j-i+1)%2==1)
                {
                    sum+=temp_sum;
                }
            }
        }
        return sum;
    }
};
